function applyFilters(entries,f,company){

  return entries.filter(e=>{

    if(company&&e.company!==company)return false

    if(f.from&&e.date<f.from)return false

    if(f.to&&e.date>f.to)return false

    if(f.company&&e.company!==f.company)return false

    if(f.supplier&&!e.supplier.toLowerCase().includes(f.supplier.toLowerCase()))return false

    if(f.obs&&!e.obs.toLowerCase().includes(f.obs.toLowerCase()))return false

    return true

  }).sort((a,b)=>b.date.localeCompare(a.date))

}

 

function calcMonthly(entries){

  const map={}

  entries.forEach(e=>{

    const m=e.date.slice(0,7)

    if(!map[m])map[m]={month:monthLabel(m),income:0,expense:0}

    if(e.type==='Entrada')map[m].income+=e.value

    else map[m].expense+=e.value

  })

  let acc=0

  return Object.entries(map).sort(([a],[b])=>a.localeCompare(b)).map(([,v])=>{

    acc+=v.income-v.expense

    return{...v,balance:acc}

  })

}

  const balance=all.reduce((s,e)=>s+(e.type==='Entrada'?e.value:-e.value),0)

  const result=income-expense

  const filtered=applyFilters(all,filters,null)

  const monthly=calcMonthly(all)

  return(

    <div>

function Reports({entries,toast}){

  const[period,setPeriod]=useState({from:'',to:''})

  const set=(k,v)=>setPeriod(f=>({...f,[k]:v}))

 

  function filtered(){

    return entries.filter(e=>{

      if(period.from&&e.date<period.from)return false

      if(period.to&&e.date>period.to)return false

      return true

    })

  }

 

  function cashflow(es){

    const map={}

    es.forEach(e=>{

      const m=e.date.slice(0,7)

      if(!map[m])map[m]={month:m,income:0,expense:0}

      if(e.type==='Entrada')map[m].income+=e.value

      else map[m].expense+=e.value

    })

    return Object.entries(map).sort(([a],[b])=>a.localeCompare(b)).map(([m,v])=>({...v,result:v.income-v.expense}))

  }

 

  function exportXLSX(){

    const es=filtered()

    if(!es.length){toast('Nenhum dado para exportar.','error');return}

    const wb=XLSX.utils.book_new()

    // Sheet 1: All

    const rows=es.map(e=>({Data:fmtDate(e.date),Tipo:e.type,Empresa:e.company,'Fornecedor/Cliente':e.supplier,'Valor (R$)':e.value,Observações:e.obs}))

    XLSX.utils.book_append_sheet(wb,XLSX.utils.json_to_sheet(rows),'Todos os Lançamentos')

    // Sheet 2: Consolidated CF

    const cf=cashflow(es)

    const cfRows=cf.map(r=>({Mês:monthLabel(r.month),'Entradas (R$)':r.income.toFixed(2),'Saídas (R$)':r.expense.toFixed(2),'Resultado (R$)':r.result.toFixed(2)}))

    XLSX.utils.book_append_sheet(wb,XLSX.utils.json_to_sheet(cfRows),'Fluxo Consolidado')

    // Sheet per company

    COMPANIES.forEach(c=>{

      const ce=es.filter(e=>e.company===c)

      if(!ce.length)return

      const cr=ce.map(e=>({Data:fmtDate(e.date),Tipo:e.type,'Fornecedor/Cliente':e.supplier,'Valor (R$)':e.value,Observações:e.obs}))

      XLSX.utils.book_append_sheet(wb,XLSX.utils.json_to_sheet(cr),c.slice(0,31))

    })

    XLSX.writeFile(wb,'relatorio_bcg.xlsx')

    toast('Exportado para Excel com sucesso!','success')

  }

 

  function exportBackup(){

    const data=JSON.stringify({entries,exportedAt:new Date().toISOString(),version:1},null,2)

    const blob=new Blob([data],{type:'application/json'})

    const a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download='backup_bcg_'+today()+'.json';a.click()

    toast('Backup exportado!','success')

  }

 

  const cf=cashflow(filtered())

  const companyCf=COMPANIES.map(c=>({company:c,cf:cashflow(filtered().filter(e=>e.company===c))}))

 

  return(

    <div>

      <div className="card" style={{marginBottom:16}}>

        <div style={{display:'flex',gap:12,alignItems:'flex-end',flexWrap:'wrap'}}>

          <div className="field"><label>Período início</label><input type="date" value={period.from} onChange={e=>set('from',e.target.value)}/></div>

          <div className="field"><label>Período fim</label><input type="date" value={period.to} onChange={e=>set('to',e.target.value)}/></div>

          <button className="btn btn-success" onClick={exportXLSX}>Exportar Excel</button>

          <button className="btn btn-ghost" onClick={exportBackup}>Exportar Backup</button>

        </div>

      </div>

 

      <div style={{marginBottom:20}}>

        <div className="section-title">Fluxo de Caixa Consolidado</div>

        <div className="table-wrap">

          <table>

            <thead><tr><th>Mês</th><th style={{textAlign:'right'}}>Entradas</th><th style={{textAlign:'right'}}>Saídas</th><th style={{textAlign:'right'}}>Resultado</th></tr></thead>

            <tbody>

              {cf.length===0?<tr><td colSpan={4} style={{textAlign:'center',color:'var(--text3)',padding:24}}>Nenhum dado no período</td></tr>:cf.map(r=>(

                <tr key={r.month}>

                  <td style={{fontWeight:500}}>{monthLabel(r.month)}</td>

                  <td className="value-cell positive" style={{textAlign:'right'}}>{fmt(r.income)}</td>

                  <td className="value-cell negative" style={{textAlign:'right'}}>{fmt(r.expense)}</td>

                  <td className={`value-cell ${r.result>=0?'positive':'negative'}`} style={{textAlign:'right',fontWeight:600}}>{fmt(r.result)}</td>

                </tr>

              ))}

            </tbody>

          </table>

        </div>

      </div>

 

      <div className="section-title">Comparativo por Empresa</div>

      <div className="table-wrap">

        <table>

          <thead><tr><th>Empresa</th><th>Mês</th><th style={{textAlign:'right'}}>Entradas</th><th style={{textAlign:'right'}}>Saídas</th><th style={{textAlign:'right'}}>Resultado</th></tr></thead>

          <tbody>

            {companyCf.flatMap(({company,cf})=>cf.map(r=>(

              <tr key={company+r.month}>

                <td><CoBadge c={company}/></td>

                <td style={{color:'var(--text2)'}}>{monthLabel(r.month)}</td>

                <td className="value-cell positive" style={{textAlign:'right'}}>{fmt(r.income)}</td>

                <td className="value-cell negative" style={{textAlign:'right'}}>{fmt(r.expense)}</td>

                <td className={`value-cell ${r.result>=0?'positive':'negative'}`} style={{textAlign:'right',fontWeight:600}}>{fmt(r.result)}</td>

              </tr>

            )))}

            {companyCf.every(({cf})=>cf.length===0)&&<tr><td colSpan={5} style={{textAlign:'center',color:'var(--text3)',padding:24}}>Nenhum dado no período</td></tr>}

          </tbody>

        </table>

      </div>

    </div>

  )

}

 

function ImportModal({open,onClose,onImport}){

  const[text,setText]=useState('')

  const[err,setErr]=useState('')

  const fileRef=useRef(null)

 

  function tryImport(){

    try{

      const d=JSON.parse(text)

      if(!Array.isArray(d.entries))throw new Error('Formato inválido')

      onImport(d.entries)

      onClose()

      setText('')

    }catch(e){setErr('JSON inválido ou formato incorreto.')}

  }

 

  function handleFile(e){

    const f=e.target.files[0];if(!f)return

    const r=new FileReader()

    r.onload=ev=>setText(ev.target.result)

    r.readAsText(f)

  }

 

  return(

    <Modal open={open} onClose={onClose} title="Importar Backup">

      <div style={{marginBottom:12}}>

        <button className="btn btn-ghost btn-sm" onClick={()=>fileRef.current.click()}>Carregar arquivo .json</button>

        <input ref={fileRef} type="file" accept=".json" style={{display:'none'}} onChange={handleFile}/>

      </div>

      <div className="field" style={{marginBottom:12}}>

        <label>Ou cole o conteúdo JSON:</label>

        <textarea rows={7} value={text} onChange={e=>{setText(e.target.value);setErr('')}} style={{fontFamily:'monospace',fontSize:11,padding:10,border:'1px solid var(--border)',borderRadius:8,width:'100%',marginTop:4}}/>

      </div>

      {err&&<div style={{color:'var(--red)',fontSize:12,marginBottom:10}}>{err}</div>}

      <div style={{display:'flex',gap:8,justifyContent:'flex-end'}}>

        <button className="btn btn-ghost" onClick={onClose}>Cancelar</button>

        <button className="btn btn-primary" onClick={tryImport}>Importar</button>

      </div>

    </Modal>

  )

}
